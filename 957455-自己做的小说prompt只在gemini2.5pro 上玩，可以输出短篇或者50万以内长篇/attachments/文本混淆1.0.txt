  function sanitizeHTML(html){
    const parser = new DOMParser();
    const doc = parser.parseFromString(`<div>${html}</div>`, 'text/html');
    const root = doc.body.firstChild;
    const walker = doc.createTreeWalker(root, NodeFilter.SHOW_ELEMENT, null);
    const toRemove = [];
    while (walker.nextNode()){
      const el = walker.currentNode;
      if (!ALLOWED_TAGS.has(el.tagName)) { toRemove.push(el); continue; }
      // 清理属性
      const allow = ALLOWED_ATTRS[el.tagName] || new Set();
      // 避免内联事件/样式
      [...el.attributes].forEach(attr => {
        const n = attr.name.toLowerCase();
        if (!allow.has(attr.name)) el.removeAttribute(attr.name);
        if (n.startsWith('on') || n === 'style') el.removeAttribute(attr.name);
        if (el.tagName === 'A' && attr.name === 'href' && !/^https?:/i.test(attr.value)) el.removeAttribute('href');
        if (el.tagName === 'A') { el.setAttribute('target','_blank'); el.setAttribute('rel','noopener noreferrer'); }
      });
    }
    toRemove.forEach(el => el.replaceWith(...el.childNodes));
    return root.innerHTML;
  }

      .reduce((acc, cur, idx, arr) => {
        if (/^[。！？!?\.]\s*$/.test(cur)) return acc;
        const tail = arr[idx+1] && /^[。！？!?\.]\s*$/.test(arr[idx+1]) ? arr[idx+1].trim() : '';
        acc.push(tail ? cur.trim() + tail : cur.trim());
        return acc;
      }, [])
      .filter(Boolean);
  }

  function nounsAndPronouns(text) {
    const found = new Set();
    const words = text.split(/(\s+|[,，。\.\-!！\?？、；;：:])/).filter(w => w && !/^\s+$/.test(w));
    for (const w of words) {
      const clean = w.replace(/[^\u4e00-\u9fffA-Za-z0-9]/g, '');
      if (!clean) continue;
      if (PRONOUNS_CN.includes(clean)) found.add(clean);
      if (/^[A-Z][a-z]+$/.test(clean)) found.add(clean);
    }
    return Array.from(found);
  }

  const PUNCTS = ['!','!!','!!!','...','?!','!?','??'];

  function addQuotes(text, intensity) {
    let out = text;
    const np = nounsAndPronouns(text);
    np.forEach(word => {
      if (Math.random()*100 < intensity) {
        if (/^[\u4e00-\u9fff]+$/.test(word)) {
          out = out.replace(new RegExp(word, 'g'), `「${word}」`);
        } else {
          out = out.replace(new RegExp(`\\b${word}\\b`, 'g'), `"${word}"`);
        }
      }
    });
    return out;
  }

  function punctNoise(text, intensity) {
    return text.replace(/[。.!！?？]+/g, (m) => (Math.random()*100 < intensity ? ' ' + pick(PUNCTS) + ' ' : m));
  }

  function caseNoise(text, intensity) {
    const words = text.split(/(\s+)/);
    for (let i = 0; i < words.length; i++) {
      const w = words[i];
      if (/^\s+$/.test(w)) continue;
      if (Math.random()*100 < intensity) {
        const a = w.slice(0,1).toUpperCase() + w.slice(1).toLowerCase();
        const b = w.toUpperCase();
        const c = w.toLowerCase();
        words[i] = pick([a,b,c]);
      }
    }
    return words.join(' ');
  }

  function replaceWithDict(text, dict, intensity) {
    let out = text;
    Object.keys(dict).forEach(key => {
      if (Math.random()*100 < intensity) {
        const alts = dict[key];
        const re = new RegExp(key.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
        out = out.replace(re, () => pick(alts));
      }
    });
    return out;
  }

  function shuffleClauses(text, intensity) {
    const sents = splitSentencesCNEN(text).map(s => {
      if (Math.random()*100 >= intensity) return s;
      let t = s;
      t = t.replace(/^(因为|如果)([^，。；！!？?]+)，(.*)$/,(m,p1,p2,p3)=> `${p3.replace(/^\s*/,'')}，${p1}${p2}`);
      return t;
    });
    return sents.join(' ');
  }

  function injectEmoji(text, intensity) { return text.replace(/([。.!！?？]+)(\s|$)/g, (match, p, sep) => (Math.random()*100 < intensity ? `${p} ${pick(EMOJIS)}${sep||''}` : match)); }

  function fakeMetaText(text, intensity){
    if (Math.random()*100 >= intensity) return text;
    const tags = ['#nofilter','#anonymous','#obfuscation','#mystery','#enigma','#AIwrites'];
    const authors = ['Author: anon','Author: unknown','By John Doe','By Jane Doe','Sent from my device','From: Classified','Author: Ghostwriter'];
    const t = Array.from({length: 1 + Math.floor(Math.random()*2)}, () => pick(tags)).join(' ');
    return text + `\n\n${t}  ${pick(authors)}`;
  }

  function fakeMetaHTML(html){
    const meta = document.createElement('p');
    meta.textContent = `${pick(['#nofilter','#anonymous','#obfuscation','#mystery'])}  ${pick(['Author: anon','By John Doe','Sent from my device'])}`;
    return sanitizeHTML(html + meta.outerHTML);
  }

  function applyStylePacks(text, selectedStyles, opts){
    let out = text; const markerOn = !!opts.reversible;
    for (const key of selectedStyles) {
      const pack = STYLE_PACKS[key]; if (!pack) continue; const map = pack.words || {};
      Object.keys(map).forEach(src => {
        const alts = map[src]; const re = new RegExp(src.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g');
        out = out.replace(re, () => { const rep = pick(alts); return markerOn ? `⟦S:${key}:${src}→${rep}⟧${rep}` : rep; });
      });
      const sents = splitSentencesCNEN(out);
      out = sents.map((s, i) => {
        if (!s) return s; let r = s;
        if (i === 0 && Math.random() < 0.8) r = (markerOn?`⟦O:${key}⟧`:``) + pick(pack.openers) + ' ' + r;
        if (Math.random() < 0.5) r = r + ' ' + (markerOn?`⟦C:${key}⟧`:``) + pick(pack.closers);
        return r;
      }).join(' ');
    }
    return out;
  }

  function decodeText(text){
    let out = text;
    out = out.replace(/[“”"\u300c\u300d\u300e\u300f『』「」]/g,'');
    EMOJIS.forEach(e => { out = out.split(e).join(''); });
    out = out.replace(/[!?！？]+/g,'。').replace(/\.{2,}|…+/g,'。');
    out = out.replace(/\u200b/g, '');
    const reverseMap = {};
    const addReverse = (src, arr) => arr.forEach(v => reverseMap[v] = src);
    Object.entries(COMMON).forEach(([k, arr]) => addReverse(k, arr));
    Object.entries(SYNONYMS).forEach(([k, arr]) => addReverse(k, arr));
    Object.entries(THEME).forEach(([t, m]) => Object.entries(m).forEach(([k, arr]) => addReverse(k, arr)));
    const tokens = Object.keys(reverseMap).sort((a,b)=>b.length-a.length);
    tokens.forEach(tok => { const re = new RegExp(tok.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g'); out = out.replace(re, reverseMap[tok]); });
    out = out.replace(/⟦S:([\w-]+):([^→]+)→([^⟧]+)⟧/g, (m,sty,src,rep)=>src);
    out = out.replace(/⟦O:[\w-]+⟧/g,''); out = out.replace(/⟦C:[\w-]+⟧/g,'');
    out = out.replace(/\s*#\S+.*$/m,''); out = out.replace(/\s*(作者:|By|From:|Author:).*$/m,''); out = out.replace(/\s*[～~].*$/m,'');
    return out.trim();
  }

  function decodeHTML(html){
    return transformHTMLText(html, decodeText);
  }

  function analyze(text){
    const tokens = []; const cn = text.match(/[\u4e00-\u9fff]{1,4}/g) || []; const en = text.match(/[A-Za-z0-9]{2,}/g) || [];
    tokens.push(...cn, ...en); const map = {}; tokens.forEach(t => map[t] = (map[t]||0) + 1);
    const sorted = Object.entries(map).sort((a,b)=>b[1]-a[1]).slice(0, 40);
    const max = sorted[0]?.[1] || 1; const cloud = sorted.map(([w,c]) => ({w, c, p: c/max}));
    return cloud;
  }

                unit_terms: ['AU','m/s²','Hz','nm'] }
    }
  };
  if (window.STYLE_PACKS) { Object.assign(window.STYLE_PACKS, STYLE_PACKS_EXT); }

  // 2) 指纹级 traits 应用器
  function splitSentencesCNEN_forTraits(text){
    return (text || '')
      .replace(/[\r\n]+/g, '\n')
      .split(/([。！？!?\.](?:\s|$))/)
      .reduce((acc, cur, i)=>{
        if (i % 2 === 0) acc.push(cur);
        else acc[acc.length-1] += cur;
        return acc;
      }, []);
  }
  function applyStyleTraits(text, traits){
    let out = text || '';

    // 标点风格
    if (traits.punctuation === 'ellipsis') {
      out = out.replace(/([。！!?])\s*/g, '…… ');
    } else if (traits.punctuation === 'emdash') {
      out = out.replace(/，([^，。！？!?]{2,8})。/g, '，$1——');
    } else if (traits.punctuation === 'staccato') {
      out = out.replace(/，/g, '。').replace(/。。+/g,'。');
    } else if (traits.punctuation === 'technical') {
      out = out.replace(/大约|大概|差不多/g, '≈');
    } else if (traits.punctuation === 'classical') {
      out = out.replace(/的/g,'之');
    }

    // 对话占比
    if (traits.dialogue_ratio){
      const sents = splitSentencesCNEN_forTraits(out);
      const tgt = Math.floor(sents.length * traits.dialogue_ratio);
      for (let i=0, c=0; i<sents.length && c<tgt; i++){
        if (!/^“.*”$/.test(sents[i]) && sents[i].replace(/\s/g,'').length>=6){
          sents[i] = `“${sents[i].replace(/[。.!！？?]$/, '')}。”`;
          c++;
        }
      }
      out = sents.join(' ');
    }

    // 悬念尾巴
    if (traits.cliffhanger && Math.random() < traits.cliffhanger){
      out = out.replace(/([。.!！？?])\s*$/, ' ——（未完）$1');
    }

    // honorifics / 术语注入
    if (traits.honorifics){
      const h = traits.honorifics[Math.floor(Math.random()*traits.honorifics.length)];
      out = out.replace(/你/g, `${h}`);
    }
    if (traits.ui_terms){
      out += `\\n\\n【Tips】${traits.ui_terms.slice(0,3).join(' / ')}`;
    }
    if (traits.system_prefix){
      out = traits.system_prefix[0] + ' ' + out;
    }
    if (traits.unit_terms){
      out = out.replace(/(\\d+)\\s*(小时|年|米)/g, (m,n,u)=>{
        const units = ['小时','年','米']; const short = ['h','yr','m'][units.indexOf(u)]; return `${n}${short}`;
      });
    }
    return out;
  }

  // 3) 重写 applyStylePacks：合并 DOM 选择 & 应用 traits
  if (window.applyStylePacks){
    const __origApply = window.applyStylePacks;
    window.applyStylePacks = function(text, selectedStyles, opts){
      const domSelected = Array.from(document.querySelectorAll('#styles .chip.active')).map(e=>e.dataset.style);
      const merged = Array.from(new Set([...(selectedStyles||[]), ...domSelected]));
      let out = __origApply.call(this, text, merged, opts || {});
      try {
        merged.forEach(k => {
          const p = window.STYLE_PACKS && window.STYLE_PACKS[k];
          if (p && p.traits) out = applyStyleTraits(out, p.traits);
        });
      } catch (e) { /* noop */ }
      return out;
    };
  }

  // 4) 动态渲染样式 chips（覆盖原有静态 HTML）
  function renderStyleChips(){
    const wrap = document.getElementById('styles'); if (!wrap || !window.STYLE_PACKS) return;
    const packs = window.STYLE_PACKS;
    wrap.innerHTML = Object.keys(packs).map(k => (
      `<div class="chip" data-style="${k}">${packs[k].name || k}</div>`
    )).join('');
    wrap.querySelectorAll('.chip').forEach(chip=>{
      chip.addEventListener('click', ()=>{
        chip.classList.toggle('active');
      });
    });
  }

  if (document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', renderStyleChips);
  } else {
    renderStyleChips();
  }
  // 可选：提示
  try { (window.toast || console.log)('已载入 2.2 扩展风格包'); } catch(e){}
})();
</script>
